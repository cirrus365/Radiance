{{/* tiles/link.html — a link moodboard tile. Parameters (single dictionary): page (page, required): the owning page; named in build warnings. tile (map, required): title (string, required): the link text; without it the tile is skipped with a build warning. description (string, optional): short supporting text. href (string, required): a site-absolute path (resolved through partials/resolve-href.html so subpath deployments work) or an absolute external URL; without it the tile is skipped with a build warning. span is accepted for a uniform signature and unused here. */}} {{- $page := .page }} {{- $tile := .tile }} {{- $href := "" }} {{- if isset $tile "href" }} {{- with $tile.href }} {{- if eq (printf "%T" .) "string" }}{{ $href = . }}{{ end }} {{- end }} {{- end }} {{- if not $tile.title }} {{- warnf "radiance: link tile on page %s has no title; the tile is skipped." $page.RelPermalink }} {{- else if not $href }} {{- warnf "radiance: link tile %q on page %s has no usable href; the tile is skipped." $tile.title $page.RelPermalink }} {{- else }} {{ $tile.title }} {{- with $tile.description }} {{ . }} {{- end }} {{- end }}
