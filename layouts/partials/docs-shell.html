{{/* docs-shell.html — the handbook layout shell shared by every chapter. Parameters (single dictionary): page (page, required): the current chapter. content (safe HTML, required): the chapter body, already rendered. Returns the chapter's window content as safe HTML: a sidebar listing every chapter (wide screens), the chapter body and Previous / Next links at the foot. Below 1024px the sidebar is replaced by a native
contents control; both representations are in the document and only one renders at a time (no script required). */}} {{- $current := .page }} {{- $nav := partial "docs-nav.html" (dict "current" $current) -}} {{- $pages := where site.RegularPages "Section" "docs" }} {{- $prev := "" }} {{- $next := "" }} {{- range $index, $page := $pages }} {{- if eq $page.RelPermalink $current.RelPermalink }} {{- if gt $index 0 }}{{ $prev = index $pages (sub $index 1) }}{{ end }} {{- if lt (add $index 1) (len $pages) }}{{ $next = index $pages (add $index 1) }}{{ end }} {{- end }} {{- end }} {{- $pagination := "" }} {{- with $prev }} {{- $pagination = printf "%sPrevious%s" $pagination .RelPermalink .Title }} {{- end }} {{- with $next }} {{- $pagination = printf "%sNext%s" $pagination .RelPermalink .Title }} {{- end }} {{- $pagination = printf "
%s
" $pagination }} {{- $article := printf "%s%s" .content $pagination | safeHTML }} {{- return (printf "
Contents
%s
%s
%s
" $nav $nav $article | safeHTML) }}
