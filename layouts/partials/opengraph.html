{{/* opengraph.html — Open Graph and Twitter card metadata. Parameters: page context (no dictionary). Uses page data: title, description (falling back to a plain-text summary truncated to 200 characters), the page kind, the publish date and the optional "cover" page-bundle resource. A raster cover is resized to a 1200px-wide JPEG and emitted as an absolute image URL; an SVG cover is passed through. Without a cover no image URL is emitted. */}} {{- $page := . }} {{- $description := $page.Description | default ($page.Summary | plainify | truncate 200) }} {{- $cover := "" }} {{- with $page.Params.cover }} {{- if eq (printf "%T" .) "string" }}{{ $cover = . }}{{ end }} {{- end }} {{- $image := "" }} {{- with $cover }} {{- with $page.Resources.Get . }} {{- if ne .MediaType.Type "image/svg+xml" }} {{- $image = (.Resize "1200x jpg q85").Permalink }} {{- else }} {{- $image = .Permalink }} {{- end }} {{- end }} {{- end }} {{- with $description }} {{- end }} {{- with $image }} {{- else }} {{- end }} {{- with $page.Date }} {{- if $page.IsPage }} {{- end }} {{- end }}
